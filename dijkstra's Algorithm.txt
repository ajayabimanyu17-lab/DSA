import heapq

def dijkstra(graph, start):
    dist = {node: float('inf') for node in graph}
    dist[start] = 0

    prev = {node: None for node in graph}
    pq = [(0, start)]
    visited = set()

    print(f"\nDijkstra's Algorithm from '{start}':")
    print(f"{'Step':<6} {'Node':<8} {'Distance':<12} {'Updated Distances'}")

    step = 1

    while pq:
        d, u = heapq.heappop(pq)

        if u in visited:
            continue

        visited.add(u)
        updates = []

        for v, w in graph[u].items():
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                prev[v] = u
                heapq.heappush(pq, (dist[v], v))
                updates.append(f"{v}={dist[v]}")

        print(
            f"{step:<6} {u:<8} {d:<12} "
            f"{', '.join(updates) if updates else '-'}"
        )

        step += 1

    print("\nShortest Distances from A:")

    for node, d in dist.items():
        path = []
        curr = node

        while curr:
            path.append(curr)
            curr = prev[curr]

        path.reverse()

        print(
            f"A -> {node}: {d} "
            f"(Path: {' -> '.join(path)})"
        )


# Main Program
graph = {
    'A': {'B': 4, 'C': 2},
    'B': {'D': 5},
    'C': {'B': 1, 'D': 8},
    'D': {'E': 6},
    'E': {}
}

dijkstra(graph, 'A')
